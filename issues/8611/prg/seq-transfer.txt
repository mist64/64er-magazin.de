 1016 ^      jsr chckom     ;"Auf Komma testen

 1460 ;"RS 232 Schnittstelle oeffnen
 1470 ;"4800 Baud, 8 Datenbits, 1 Stopbit

 1485 ;"Full Duplex

 1500 ^      ldx #2         ;"Geraetenummer: 2
 1510 ^      ldy #3         ;"Sekundaeradresse: 3
 1520 ^      jsr setlfs     ;"setzen
 1530 ^      lda #4         ;"Werte fuer

 1610 ^      dex
 1620 ^      bpl baudl9
 1650 ;
 1660 ^      ldx #1         ;"File als Ausgabefile
 1670 ^      jsr chkout     ;"deklarieren
 1680 ^      ldx #<start    ;"Startadresse fuer
 1690 ^      ldy #>start    ;"das SEQ-File als Zeiger
 1700 ^      stx hpt        ;"merken
 1710 ^      sty hpt + 1
 1712 ^      lda 56334      ;"Timerirq sperren
 1714 ^      and #254
 1716 ^      sta 56334
 1720 ;
 1730 baudl2 ldy #0         ;"Aktuelles Zeichen
 1740 ^      lda (hpt),y    ;"holen
 1750 ^      bit mconv      ;"Zeichen konvertieren?
 1760 ^      bmi baud51     ;"NEIN: => weiter

 1780 ^      bcc baudw5     ;"nach ASCII
 1790 ^      cmp #128
 1800 ^      bcs baud41
 1810 ^      ora #%00100000
 1820 ^      bne baudw5

 1890 baudl3 lda enable     ;"Ist das vorherige Zeichen

 1940 ^      bne baudw6     ;"Zeichen um eins
