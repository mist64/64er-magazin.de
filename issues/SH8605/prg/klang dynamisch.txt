  110 rem klangeffekte mit

  150 read a,d,su,r,c,p,f,g,n,m
  160 poke s+5 ,16*a +d
  170 poke s+6 ,16*su+r

  190 poke s+3 ,p/256
  200 poke s+23,0     :rem fr
  210 poke s+24,15    :rem ml

  240 : f1=f
  250 : poke s+4,c or 1
  260 : for j=1 to n
  270 :   poke s,f1 and 255
  280 :   poke s+1,f1/256
  290 :   f1=f1*g
  300 : next j
  310 : poke s+4,c
  320 next i
  330 goto 220
